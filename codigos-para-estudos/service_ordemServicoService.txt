// Gera número sequencial da OS baseado no ano atual
export const generateOsNumber = async (): Promise<string> => {
  const currentYear = new Date().getFullYear();
  const yearPrefix = currentYear.toString();
  
  try {
    // Busca a última OS do ano para gerar próximo número
    const { data, error } = await supabase
      .from('bd_ordens_servico')
      .select('numero_os')
      .like('numero_os', `${yearPrefix}%`)
      .order('numero_os', { ascending: false })
      .limit(1);

    if (error) throw error;

    let nextNumber = 1;
    if (data && data.length > 0) {
      const lastNumber = data[0].numero_os;
      const lastSequence = parseInt(lastNumber.replace(yearPrefix, ''));
      nextNumber = lastSequence + 1;
    }

    return `${yearPrefix}${nextNumber.toString().padStart(4, '0')}`;
  } catch (error) {
    console.error('Erro ao gerar número da OS:', error);
    throw new Error('Erro ao gerar número da OS');
  }
};

// Busca chamados abertos com filtros e joins
export const fetchOpenChamados = async (filters: OsFilterParams = {}) => {
  try {
    console.log('🔍 Buscando chamados abertos com filtros:', filters);
    
    let query = supabase
      .from('bd_chamados_os')
      .select(`
        *,
        solicitante:profiles!bd_chamados_os_solicitante_id_fkey(email, nome_completo),
        centro_custo:bd_centro_custo!bd_chamados_os_centro_custo_id_fkey(codigo_centro_custo, nome_centro_custo),
        caminhao_equipamento:bd_veiculos!bd_chamados_os_caminhao_equipamento_id_fkey(id, placa, tipo_veiculo, marca, modelo)
      `)
      .in('status', ['Aberto', 'Em Andamento', 'Aguardando Peças']);

    // Aplicação de filtros
    if (filters.numero_os) {
      query = query.ilike('numero_chamado', `%${filters.numero_os}%`);
    }
    
    if (filters.caminhao_equipamento_id) {
      query = query.eq('caminhao_equipamento_id', filters.caminhao_equipamento_id);
    }
    
    if (filters.data_inicio && filters.data_fim) {
      query = query.gte('data_abertura', filters.data_inicio)
                  .lte('data_abertura', filters.data_fim);
    }

    const { data, error } = await query.order('data_abertura', { ascending: false });
    
    if (error) throw error;
    
    console.log('✅ Chamados carregados:', data?.length || 0);
    return data || [];
  } catch (error) {
    console.error('❌ Erro ao buscar chamados:', error);
    throw error;
  }
};

// ⚠️ FUNÇÃO PROBLEMÁTICA - Atualiza OS existente
export const updateOrdemServico = async (
  id: string, 
  osData: Partial<OrdemServico>, 
  currentUserId: string, 
  motivo: string
): Promise<OrdemServico> => {
  try {
    console.log('📝 Atualizando OS:', id);
    console.log('📋 Dados recebidos:', osData);

    // Limpeza e validação dos dados
    const cleanedData = { ...osData };
    
    // Remove campos undefined, null ou vazios
    Object.keys(cleanedData).forEach(key => {
      const value = cleanedData[key as keyof OrdemServico];
      if (value === undefined || value === null || value === '') {
        delete cleanedData[key as keyof OrdemServico];
      }
    });

    // Remove campos de relacionamento que não devem ser atualizados
    delete cleanedData.caminhao_equipamento;
    delete cleanedData.executado_por;
    delete cleanedData.encerrado_por;
    delete cleanedData.centro_custo;
    delete cleanedData.solicitante;

    // Conversão de campos numéricos
    if (cleanedData.valor_servico !== undefined) {
      cleanedData.valor_servico = parseFloat(cleanedData.valor_servico.toString()) || 0;
    }

    // Atualização no banco
    const { data, error } = await supabase
      .from('bd_ordens_servico')
      .update({
        ...cleanedData,
        updated_at: new Date().toISOString()
      })
      .eq('id', id)
      .select(`
        *,
        caminhao_equipamento:bd_veiculos!bd_ordens_servico_caminhao_equipamento_id_fkey(
          id, placa, tipo_veiculo, marca, modelo
        ),
        executado_por:profiles!bd_ordens_servico_executado_por_id_fkey(
          id, email, nome_completo
        ),
        encerrado_por:profiles!bd_ordens_servico_encerrado_por_id_fkey(
          id, email, nome_completo
        )
      `)
      .single();

    if (error) {
      console.error('❌ Erro na atualização:', error);
      throw error;
    }

    if (!data) {
      throw new Error('Nenhum dado retornado após atualização');
    }

    console.log('✅ OS atualizada com sucesso:', data.numero_os);
    return data;
  } catch (error) {
    console.error('❌ Erro ao atualizar OS:', error);
    throw error;
  }
};

// ⚠️ FUNÇÃO CRÍTICA PROBLEMÁTICA - Salva material
export const saveMaterial = async (material: Partial<Material>): Promise<Material> => {
  try {
    console.log('💾 Salvando material:', material);
    
    // ⚠️ VALIDAÇÃO CRÍTICA
    if (!material.os_id) {
      console.error('❌ OS ID não fornecido para o material');
      throw new Error('OS ID é obrigatório para salvar material');
    }

    // Validações básicas
    if (!material.descricao_material?.trim()) {
      throw new Error('Descrição do material é obrigatória');
    }

    if (!material.quantidade || material.quantidade <= 0) {
      throw new Error('Quantidade deve ser maior que zero');
    }

    // Preparação dos dados
    const materialData = {
      os_id: material.os_id,
      descricao_material: material.descricao_material.trim(),
      quantidade: parseFloat(material.quantidade.toString()),
      valor_unitario: parseFloat(material.valor_unitario?.toString() || '0'),
      valor_total: parseFloat(material.quantidade.toString()) * parseFloat(material.valor_unitario?.toString() || '0')
    };

    console.log('📊 Dados preparados:', materialData);

    let result;
    
    if (material.id) {
      // ⚠️ ATUALIZAÇÃO DE MATERIAL EXISTENTE
      console.log('📝 Atualizando material existente:', material.id);
      
      const { data, error } = await supabase
        .from('bd_os_materiais_utilizados')
        .update({
          ...materialData,
          updated_at: new Date().toISOString()
        })
        .eq('id', material.id)
        .select('*')
        .single();

      if (error) throw error;
      result = data;
    } else {
      // ⚠️ CRIAÇÃO DE NOVO MATERIAL
      console.log('🆕 Criando novo material');
      
      const { data, error } = await supabase
        .from('bd_os_materiais_utilizados')
        .insert({
          ...materialData,
          created_at: new Date().toISOString(),
          updated_at: new Date().toISOString()
        })
        .select('*')
        .single();

      if (error) throw error;
      result = data;
    }

    if (!result) {
      throw new Error('Nenhum dado retornado após salvamento do material');
    }

    console.log('✅ Material salvo com sucesso:', result.id);
    return result;
  } catch (error) {
    console.error('❌ Erro ao salvar material:', error);
    throw error;
  }
};
